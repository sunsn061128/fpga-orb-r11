cmake_minimum_required(VERSION 3.10)
project(orb_frontend_golden LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(orb_golden golden.cpp)
target_compile_options(orb_golden PRIVATE -Wall -Wextra -Wconversion)
add_executable(orb_golden_cli main.cpp)
target_link_libraries(orb_golden_cli orb_golden)
add_executable(orb_golden_unit tests/unit.cpp)
target_link_libraries(orb_golden_unit orb_golden)
enable_testing()
add_test(NAME integer_unit COMMAND orb_golden_unit)
